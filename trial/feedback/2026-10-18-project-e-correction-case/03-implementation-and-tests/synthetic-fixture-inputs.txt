def _record_rewrite_problem_transaction(
    context: BidAgentContext,
    *problems: dict[str, object],
) -> dict[str, object]:
    ctx = SimpleNamespace(context=context)
    model_problems: list[dict[str, object]] = []
    for problem in problems:
        projected = dict(problem)
        leaf_id = str(projected.pop("_testLeafId"))
        body_payload = sdk_tools.store_read_latest_generated_bodies(context.cid, [leaf_id])
        body_node = body_payload["nodes"][0]
        matches, gap = sdk_tools.issue_rewrite_problem_candidate_handles(
            context,
            candidates=[
                {
                    "leafNodeId": leaf_id,
                    "latestVersion": body_node["latestVersion"],
                    "leafTitle": "正文候选",
                    "contentExcerpt": body_node["content"],
                }
            ],
            identity_resolver=lambda leaf_ids: sdk_tools._rewrite_current_identities(ctx, leaf_ids, fresh=True),
            body_reader=lambda leaf_ids: sdk_tools.store_read_latest_generated_bodies(context.cid, leaf_ids),
        )
        assert gap is None
        _observe_test_body_candidates(context, matches)
        projected["candidateHandle"] = matches[0]["candidateHandle"]
        model_problems.append(projected)
    problem_set_payload: dict[str, object] = {"problems": model_problems}
    outcome = _sdk_tool_function(sdk_tools.record_rewrite_problem_set)(
        ctx,
        problem_set_payload,
    )
    if inspect.iscoroutine(outcome):
        outcome = asyncio.run(outcome)
    return outcome
